Packiko Video SDK — vanilla walkthrough

ทีละขั้นด้วย JavaScript ล้วน ไม่ต้องมี bundler/React — เปิดกล่องไล่จากบนลงล่าง แล้วดูผลจริงใน Event log ล่างสุด

ขั้น 1 · ตั้งค่า SDK

ใช้แค่ publishable key (pk_...) — ปลอดภัยในเบราว์เซอร์ ไม่มี secret

<script src="https://sdk-uat.packiko.com/video/v0.3.0/index.global.js"></script>
const { createRecorder, createPlayer, createUploadQueue } = PackikoVideo;
const config = { apiBaseUrl, publicKey };
ขั้น 2 · อัดวิดีโอ + อัปโหลด

กล้องเปิดตอนกางกล่องนี้ · กด Stop แล้วไฟล์อัปโหลดตรงขึ้น storage (ไม่ผ่าน server ของคุณ) ได้ videoId กลับมา

const cap = await createRecorder(config).capture();  // ขอกล้อง + preview
cap.start();                                          // เริ่มอัด
const blob = await cap.stop();                        // ได้ไฟล์
const up = createRecorder(config).upload(blob, { orderRef });
const { videoId } = await up.promise;                 // token → PUT → confirm

เอา videoId ไปขอลิงก์เล่น — ถ้าคลิปยัง transcode ไม่เสร็จ SDK รอให้จนพร้อม

const player = createPlayer(config);
const { url } = await player.resolvePlaybackUrl(videoId);

state: idle

ขั้น 4 · คิวถาวร + ผูกเอกสารทีหลัง

enqueue() เก็บคลิปลงเครื่อง "ก่อน" ยิง network — refresh/ปิดแท็บ/เน็ตหลุด คลิปไม่หาย · job จะจอดรอจน "เอกสาร" พร้อม (releaseWhen) แล้วค่อยผูก (attach)

const queue = createUploadQueue(config, {
  releaseWhen: (job) => documentReady(job.context),   // false = จอดรอ ไม่ใช่ fail
  attach: (job) => bindClip(job.context, job.videoId), // ต้อง idempotent
});
await queue.enqueue({ blob, context, orderRef });      // durable ทันที
await queue.nudge();                                   // เอกสารพร้อมแล้ว เช็คเดี๋ยวนี้
